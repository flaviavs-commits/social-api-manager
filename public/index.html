// Rota de callback no servidor (src/routes/oauth.js) router.get('/meta/callback', async (req, res) => { const { code, state } = req.query; // Troca code por access_token na API do Facebook const tokenRes = await fetch('https://graph.facebook.com/oauth/access_token', { method: 'POST', body: new URLSearchParams({ client_id: process.env.META_APP_ID, client_secret: process.env.META_APP_SECRET, redirect_uri: process.env.META_REDIRECT_URI, code }) }); const { access_token, expires_in } = await tokenRes.json(); // Salvar em data/db.json via lowdb db.get('tokens').push({ access_token, expires_in }).write(); res.redirect('/?connected=true'); });
